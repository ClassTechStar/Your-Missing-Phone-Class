\bookchapter{Annexe B : Petit guide d'exercices pratiques}

\begin{itemize}
  \item \textbf{Exercices de base}:Nettoyez le stockage du smartphone : trouvez et supprimez les vieilles vidéos « oubliées », les captures d’écran en double et les applis inutilisées depuis longtemps.
  \item \textbf{Exercices logiciels}:Configurez le nettoyage de stockage WeChat, le chemin est généralement [Moi] - [Paramètres] - [Général] - [Espace de stockage].
  \item \textbf{Exercices avancés}:Vérifiez l’état de la batterie, sur iPhone dans [Réglages] - [Batterie] - [Santé de la batterie].
  \item \textbf{Astuces générales}:En cas de problème, redémarrez d’abord ! C’est la clé universelle qui règle plus de la moitié des petits soucis de smartphone.
\end{itemize}
