\bookchapter{Lampiran D: Di Mana Mencari Sumber Daya Tepercaya?}

\booksection{Dukungan resmi adalah pilihan utama:}

\begin{itemize}
  \item \textbf{Pengguna Apple}: apple.com/cn dan iCloud.com.
  \item \textbf{Pengguna Android}: Situs resmi merek ponsel memiliki bagian "Layanan dan Dukungan".
  \item \textbf{Alat umum}: Apple gunakan iTunes, Huawei gunakan Huawei Phone Assistant, Xiaomi gunakan Xiaomi Phone Switch.
\end{itemize}

\booksection{Situs dan blogger praktis:}

\begin{itemize}
  \item \textbf{Mencari tips}: Cari "tips [model ponsel Anda]" di Bilibili.
  \item \textbf{Menonton ulasan}: Tonton video ulasan dari beberapa blogger yang berbeda.
  \item \textbf{Belajar fotografi}: Ikuti blogger tutorial fotografi ponsel.
\end{itemize}

\booksection{Daftar periksa keamanan satu kalimat:}

\begin{enumerate}
  \item Apakah kata sandi layar kunci sudah diatur?
  \item Apakah "Find My Phone" sudah diaktifkan?
  \item Apakah otentikasi dua faktor untuk akun penting sudah diaktifkan?
  \item Apakah Anda secara teratur mencadangkan foto dan riwayat chat WeChat?
  \item Ingat mantra anti-penipuan: jangan mudah percaya, jangan bocorkan informasi, jangan transfer uang, jangan serakah, jangan terburu-buru.
\end{enumerate}
