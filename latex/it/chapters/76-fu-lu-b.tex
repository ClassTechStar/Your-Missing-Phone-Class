\bookchapter{Appendice B: Piccoli esercizi pratici}

\begin{itemize}
  \item \textbf{Esercizio Fondamenti}: pulisci la memoria del telefono; l'obiettivo è trovare ed eliminare quei vecchi video "dimenticati", screenshot doppi e app inutilizzate da tempo.
  \item \textbf{Esercizio Software}: configura la pulizia della memoria di WeChat; il percorso di solito è [Io] - [Impostazioni] - [Generali] - [Spazio di archiviazione].
  \item \textbf{Esercizio Avanzato}: controlla lo stato di salute della batteria; su iPhone in [Impostazioni] - [Batteria] - [Salute della batteria].
  \item \textbf{Consiglio generale}: di fronte a un problema, prima riavvia! È la chiave universale per risolvere più della metà dei piccoli guai del telefono.
\end{itemize}
