\bookchapter{付録D：信頼できるリソースはどこで探す？}

\booksection{公式サポートが第一選択：}

\begin{itemize}
  \item \textbf{付録D：信頼できるリソースはどこで探す？}公式サポートが第一選択：
  \item \textbf{Appleユーザー}：apple.com/cn と iCloud.com。
  \item \textbf{Androidユーザー}：スマートフォンブランドの公式サイトに「サービスとサポート」のセクションがあります。
\end{itemize}

\booksection{実用サイトとブロガー：}

\begin{itemize}
  \item \textbf{：AppleはiTunes、HuaweiはHuaweiスマートフォンアシスタント、XiaomiはXiaomi機種変更アプリ。}実用サイトとブロガー：
  \item \textbf{テクニックを探す}：Bilibiliで「【あなたのスマートフォン型番】テクニック」を検索。
  \item \textbf{レビューを見る}：複数の異なるブロガーのレビュー動画を見る。
\end{itemize}

\begin{enumerate}
