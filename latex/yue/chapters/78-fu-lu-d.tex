\bookchapter{附錄D：靠譜資源去邊度揾？}

\booksection{官方支持係首選：}

\begin{itemize}
  \item \textbf{蘋果用戶}：apple.com/cn 同 iCloud.com。
  \item \textbf{安卓用戶}：手機品牌嘅官網都有「服務與支持」板塊。
  \item \textbf{通用工具}：蘋果用iTunes，華為用華為手機助手，小米用小米換機。
\end{itemize}

\booksection{實用網站同博主：}

\begin{itemize}
  \item \textbf{查技巧}：喺B站搜尋「【你嘅手機型號】技巧」。
  \item \textbf{睇測評}：多睇幾個唔同博主嘅測評影片。
  \item \textbf{學影相}：關注手機攝影教程博主。
\end{itemize}

\booksection{安全自查一句話清單：}

\begin{enumerate}
  \item 鎖屏密碼設咗未？
  \item 「查找我嘅手機」開咗未？
  \item 重要帳號嘅雙重認證開咗未？
  \item 定期畀相、微信聊天記錄做備份咗未？
  \item 記住防騙口訣：唔輕信、唔洩露、唔轉賬、唔貪心、唔急躁。
\end{enumerate}
