\bookchapter{Apêndice D: Onde Encontrar Recursos Confiáveis?}

\booksection{Suporte oficial é a primeira escolha:}

\begin{itemize}
  \item \textbf{Usuários Apple}: apple.com/cn e iCloud.com.
  \item \textbf{Usuários Android}: O site oficial da marca do celular tem uma seção de "Serviço e Suporte".
  \item \textbf{Ferramentas universais}: iTunes para Apple, Assistente Huawei para Huawei, Mi Mover para Xiaomi.
\end{itemize}

\booksection{Sites e criadores úteis:}

\begin{itemize}
  \item \textbf{Buscar dicas}: Pesquise no Bilibili por ""modelo do seu celular" dicas".
  \item \textbf{Ver análises}: Assista a vídeos de análise de vários criadores diferentes.
  \item \textbf{Aprender fotografia}: Siga criadores de tutoriais de fotografia mobile.
\end{itemize}

\booksection{Lista de verificação de segurança em uma frase:}

\begin{enumerate}
  \item Você definiu uma senha de bloqueio de tela?
  \item O "Buscar Meu Dispositivo" está ativado?
  \item A autenticação de dois fatores das contas importantes está ativada?
  \item Você faz backup regularmente das fotos e do histórico de conversas do WeChat?
  \item Lembre-se do mantra anti-fraude: Não acredite facilmente, não divulgue dados, não transfira dinheiro, não seja ganancioso, não se apresse.
\end{enumerate}
