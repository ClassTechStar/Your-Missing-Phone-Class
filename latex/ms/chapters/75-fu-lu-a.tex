\bookchapter{Lampiran A: Proses Unboxing Telefon Baru – Panduan “Peringkat Pengasuh”}

\booksection{“Tiga Langkah” Pemeriksaan Unboxing:}

\begin{itemize}
  \item \textbf{Cuba fungsi}:Buat panggilan untuk uji pembesar suara dan mikrofon, ambil foto untuk periksa kamera.
  \item \textbf{2. Peringatan Penting Pengaktifan \& Tetapan Awal:}Rangkaian
  \item \textbf{:Sambung ke Wi-Fi yang stabil terlebih dahulu.}Akaun
\end{itemize}

\booksection{Peringatan Penting Pengaktifan \& Tetapan Awal:}

\begin{itemize}
  \item \textbf{Tetapan asas}:Hidupkan "Cari Telefon Saya", tetapkan kata laluan skrin kunci yang kukuh.
  \item \textbf{Tetapan eSIM}:Jika telefon anda menyokong eSIM (seperti iPhone, beberapa model Android flagship), anda boleh menambah pelan eSIM dalam tetapan tanpa perlu memasukkan kad SIM fizikal. eSIM sangat mudah semasa ke luar negara, anda boleh membeli pelan eSIM pengendali tempatan secara langsung.
  \item \textbf{Pemindahan data}:Semasa memindahkan data dari telefon lama, pastikan kedua-dua telefon mempunyai cas bateri yang mencukupi. iPhone gunakan "Quick Start", Android gunakan App pemindahan telefon rasmi.
  \item \textbf{3. Pasang App – "Keselamatan Diutamakan":}Kedai rasmi diutamakan
  \item \textbf{:Apple gunakan App Store, Android gunakan kedai aplikasi bawaan telefon.}Jangan beri kebenaran sembarangan
\end{itemize}

\booksection{Pasang App – “Keselamatan Diutamakan”:}

\begin{itemize}
  \item \textbf{Lampiran B: Rujukan Latihan Amali}Latihan bab asas
  \item \textbf{:Bersihkan storan telefon, sasaran adalah mencari dan memadam video lama "lupa apa itu", tangkap layar berulang, dan App yang sudah lama tidak digunakan.}Latihan bab perisian
\end{itemize}
