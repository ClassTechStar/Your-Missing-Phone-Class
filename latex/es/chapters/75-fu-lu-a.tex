\bookchapter{Apéndice A: Flujo “nivel niñera” para desempaquetar un teléfono inteligente nuevo}

\booksection{“Trilogía” de desempaquetado y verificación:}

\begin{itemize}
  \item \textbf{Donde se guardan fotos, videos y Apps. Cuando dicen 128G o 256G, se refieren a esto.}OLED / LCD
  \item \textbf{Dos tipos de materiales de pantalla}OLED: colores más vivos y menor consumo; LCD: más cuidado visual y menor costo.
  \item \textbf{NFC}Comúnicacion inalámbrica de corto alcance
\end{itemize}

\booksection{Recordatorios importantes para la activación y configuración inicial:}

\begin{itemize}
  \item \textbf{Actualización OTA}Actualizacion del sistema inalámbrica
  \item \textbf{Esa opción "actualización del sistema" en [Configuracion] es esto.}Root
  \item \textbf{Permisos de superadministrador de Android}Tras obtenerlo puedes modificar archivos del sistema, pero conlleva riesgos.
  \item \textbf{Bloqueo BL}Bloqueo de Bootloader
  \item \textbf{Cerradura de seguridad del fabricante; debe desbloquearse para poder flashear.}NPU
\end{itemize}

\booksection{Instalar aplicaciones con “seguridad primero”:}

\begin{itemize}
  \item \textbf{Chip especializado en tareas de IA; decenas de veces más rápido que la CPU con menor consumo.}IA en dispositivo
  \item \textbf{IA que funciona localmente en el móvil}La IA se ejecuta directamente en el móvil sin necesidad de conexión; rápida y con buena privacidad.
\end{itemize}
